# ============================================================================
#  sage — командная строка движка.
#
#  Ни окна, ни OpenGL: инструмент линкует sage::engine ради проекта, сцен и
#  сериализации, но Application не создаёт. Поэтому он собирается и работает
#  на машине без графики — на сборочном сервере, по ssh, в git-хуке.
#
#  Это же и есть проверка одного из архитектурных требований: если бы системы
#  движка были завязаны на существование окна, эта цель не собралась бы.
# ============================================================================
add_executable(sage_cli main.cpp)

# Имя программы — «sage», а не «sage_cli»: под ним её набирает человек.
# Цель называется иначе, потому что sage — уже пространство имён и префикс
# половины целей проекта, и коллизия имён в CMake ничем хорошим не кончается.
set_target_properties(sage_cli PROPERTIES OUTPUT_NAME "sage")

sage_windows_manifest(sage_cli)
target_link_libraries(sage_cli PRIVATE sage::engine)
